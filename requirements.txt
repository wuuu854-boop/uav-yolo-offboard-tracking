# Runtime dependencies (top-level, deliberately loose).
# For an exact, reproducible environment run `make lock` on the target machine
# and commit the generated requirements.lock.txt; that file is what the SITL
# bring-up in docs/setup-log.md produced (numpy 1.26.4, opencv 4.10.0.84).
# CI installs requirements-dev.txt, not the lock, so a dependency change shows up
# there as well.
#
# numpy is pinned below 2.0 on purpose: ROS 2 Humble's cv_bridge and the system
# OpenCV are built against the numpy 1.x ABI. With numpy 2 the import fails with
# "_ARRAY_API not found" and then segfaults inside cv_bridge.
numpy>=1.26,<2.0
# Headless, not opencv-python: the GUI build links libGL, which CI containers and
# headless SITL machines do not have, and importing cv2 then fails with
# "libGL.so.1: cannot open shared object file". Nothing here calls imshow.
# 4.11+ requires numpy>=2, which the pin above forbids: pip then backtracks
# through every opencv release, downloading 70 MB wheels each time, and takes
# minutes to fail on a fresh Ubuntu 22.04. Keep the ceiling.
opencv-python-headless>=4.9,<4.11
onnxruntime>=1.17,<2.0
PyYAML>=6.0,<7.0
# The read-only telemetry mirror. Optional at runtime — nothing in the control
# loop imports it — but it is a runtime dependency of a shipped script, so it
# lives here rather than in requirements-dev.
pymavlink>=2.4,<3.0
